Table Account {
  _id uuid [primary key]
  username varchar(50) [unique, not null]
  password varchar(100) [not null]
  password_last_changed datetime [not null]
  password_status Entity [not null, default: 'Temporary'] // 'Temporary', 'Normal', 'Expired'
  password_expiry_time datetime  
  must_change_password bool [not null, default: true]
  last_failed_login_time timestamp
  role Entity [not null]
  site Entity
  area Entity
  account_type Entity [not null] // 'Permanent', 'WithExpirationDate'
  account_expiry_date datetime
  last_login_time timestamp
  is_active bool [not null, default: true]
  is_deleted bool [not null, default: false]
  full_name varchar(100) [unique, not null]
  // created_by varchar(100) [not null] dari event log tables
  // created_date timestamp
  // modified_by varchar(100) [not null]
  // modified_date timestamp
}

ref: Account.role > Entity._id
ref: Account.site > Entity._id
ref: Account.area > Entity._id
ref: Account.password_status > Entity._id

  assigned_operator_2 Account

  planned_end_time datetime [not null] // diisi berdasarkan hasil calc. dari estimate API. on request 
  job_priority Entity // High, Medium, Low
  job_lifecycle_state Entity // scheduled, released, running, suspended, completed, cancelled, closed.

////////// FR 5 

Table Entity {
  _id uuid [primary key]
  entity_name varchar(200) [not null]
  description string 
  entity_class varchar(200) [not null] 
  is_active boolean [default: true]
  equipment_capabilities Entity_Capability[]
  equipment_state Entity_State
}

Table Entity_Capability {
  _id uuid [primary key]
  quantity int [not null]
  time_unit varchar(20) [not null]
  capability_type varchar(100) [not null]
  effective_start datetime
  effective_end datetime
  version int [increment]
}

Table Entity_State {
  _id uuid [primary key]
  state enum('Available','Running','Down','Maintenance', 'Inactive') [not null, default: 'Inactive']
  capability_type varchar(100) [not null]
  state_start_time timestamp
  state_end_time timestamp
}

ref: Entity.equipment_capabilities < Entity_Capability._id
ref: Entity.equipment_state - Entity_State._id
